Does the volume title follow the project convention?

Note

The name ‘prefix’ contains “PLAYProject_” and the ‘postfix’ is a 5-chararacter string with all capital letters or a number.

vol_name_ok <- function(vol_df) {
  assertthat::assert_that(!is.null(vol_df))
  assertthat::has_name(vol_df, "title")
  prefix <- "PLAYProject_"
  prefix_ok <- stringr::str_detect(vol_df$title, paste0("^", prefix))
  postfix_ok <- stringr::str_detect(vol_df$title, "[A-Z1-2]{5}$")
  prefix_ok & postfix_ok
}

vol_name_ok(vol_df)

make_session_folder_prefix <- function(vol_id) {
  assertthat::is.number(vol_id)
  assertthat::assert_that(vol_id > 0)
  
  vol_df <- databraryr::get_volume_by_id(vol_id = vol_id)
  assertthat::assert_that(!is.null(vol_df))
  assertthat::has_name(vol_df, "title")
  
  paste0("PLAY_", stringr::str_extract(vol_df$title, "[A-Z1-2]{5}$"), "_")
}

folder_name_ok <- function(vol_folders, prefix = "PLAY_PENNS_") {
  assertthat::assert_that(!is.null(vol_folders))
  assertthat::has_name(vol_folders, "folder_name")
  assertthat::is.string(prefix)
  
  names <- vol_folders$folder_name
  stringr::str_detect(names, "demographics")
  
  prefix_ok <- stringr::str_detect(names, paste0("^", prefix))
  postfix_ok <- stringr::str_detect(names, "demographics$")
  prefix_ok & postfix_ok
}

session_name_suffix_ok <- function(name) {
  assertthat::is.string(name)
  
  suffix <- stringr::str_extract(name, "[0-9]{3}$")
  as.numeric(suffix) > 0
}

session_name_prefix_ok <- function(name, prefix) {
  assertthat::is.string(name)
  assertthat::is.string(prefix)
  session_prefix <- stringr::str_extract(name, paste0("PLAY_[A-Z0-9]{5}_"))
  prefix == session_prefix
}

Does the file have the proper release level?

demog_file_release_ok <- function(vol_folders, vol_id) {
  assertthat::assert_that(!is.null(vol_folders))
  assertthat::is.number(vol_id)
  assertthat::assert_that(vol_id > 0)
  
  folder_assets <- databraryr::list_folder_assets(folder_id = vol_folders$folder_id, 
                                      vol_id = vol_id)
  assertthat::assert_that(!is.null(folder_assets))
  
  demog_file <- folder_assets |>
    dplyr::filter(stringr::str_detect(asset_name, "demographics"))
  
  demog_file$asset_permission == "authorized_users"
}
